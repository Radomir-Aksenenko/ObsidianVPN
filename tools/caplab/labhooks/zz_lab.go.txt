package main

// Lab-only hooks, copied into cmd/server by tools/caplab when it builds test servers; never part
// of a release build. OBSIDIAN_CPUPROFILE + OBSIDIAN_CPUPROFILE_SECS write a CPU profile;
// OBSIDIAN_ACK_HOLD_BYTES overrides obsidian.ACKHoldBytes.

import (
	"os"
	"runtime/pprof"
	"strconv"
	"time"

	"obsidian/obsidian"
)

func init() {
	if v, err := strconv.ParseUint(os.Getenv("OBSIDIAN_ACK_HOLD_BYTES"), 10, 32); err == nil {
		obsidian.ACKHoldBytes = uint32(v)
	}
	path := os.Getenv("OBSIDIAN_CPUPROFILE")
	secs, _ := strconv.Atoi(os.Getenv("OBSIDIAN_CPUPROFILE_SECS"))
	if path == "" || secs <= 0 {
		return
	}
	f, err := os.Create(path)
	if err != nil {
		return
	}
	_ = pprof.StartCPUProfile(f)
	go func() {
		time.Sleep(time.Duration(secs) * time.Second)
		pprof.StopCPUProfile()
		f.Close()
	}()
}
